class Solution {
public:
    int maxArea(vector<int>& height) {
        int lp = 0,rp = height.size()-1;
        int maxWater=0;
        while(rp>lp){
            int W=rp-lp;
            int ht=min(height[rp],height[lp]);
            int curr_Water=W*ht;
            maxWater=max(maxWater,curr_Water);
            if(height[rp]>height[lp]){
                lp++;
            }
            else{
                rp--;
            }
        }
        return maxWater;
    }
};
