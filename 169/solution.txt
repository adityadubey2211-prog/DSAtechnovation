class Solution {
public:
    int majorityElement(vector<int>& nums) {
        int n = nums.size();
        int flag = 0;
        int ans = 0;
        for(int i = 0; i<n; i++){
            if(flag == 0){
                ans = nums[i];
            }
            if(nums[i] == ans){
                flag++;
            }
            else{
                flag --;
            }
        }
        return ans;
    }
};
