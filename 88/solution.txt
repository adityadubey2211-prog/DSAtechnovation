class Solution {
public
    void merge(vectorint& A, int m, vectorint& B, int n) {
        int index=n+m-1;
        int i = m-1;
        int j = n-1;
        while(i=0 && j=0){
            if(A[i]=B[j]){
                A[index--] = A[i--];
            }
            else{
                A[index--] = B[j--];
            }
        }
        while(j=0){
            A[index--] = B[j--];
        }
    }
};
