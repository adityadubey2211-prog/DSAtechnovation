class Solution {
public:
    bool isValidSudoku(vector<vector<char>>& board) {
        int row[9][10] = {0};
        int col[9][10] = {0};
        int box[9][10] = {0};
        for(int i = 0; i < 9; i++) {
            for(int j = 0; j < 9; j++) {
                if(board[i][j] == '.') {
                    continue;
                }
                int num = board[i][j] - '0';
                int bidx = (i / 3) * 3 + (j / 3);
                if(row[i][num] || col[j][num] || box[bidx][num]) {
                    return false;
                }
                row[i][num] = 1;
                col[j][num] = 1;
                box[bidx][num] = 1;
            }
        }
        return true;
    }
};
