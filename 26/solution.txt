class Solution {
public:
    int removeDuplicates(vector<int>& nums) {
        int n = nums.size();
        int j = 0;
        int k = 1;
        for(int i = 1; i<n; i++){
            if(nums[i] == nums[j]){
                continue;
            }
            j++;

            nums[j] = nums[i];
            
            k++;
        }
        return k;
    }
};
